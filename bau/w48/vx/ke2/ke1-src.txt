function lsd(x, y) {

function nelderMead(f, x0, step, maxEval, bounds) {

while (ev < maxEval) {

const xr = clip(c.map((v, j) => v + (v - S[n][j]))), fr = f(xr); ev++;

else if (fr < F[n - 1]) { S[n] = xr; F[n] = fr; }
else { const xc = clip(c.map((v, j) => v + .5 * (S[n][j] - v))), fc = f(xc); ev++;

}

}

function eigenSym(C, n) {
const a = C.map(r => r.slice()), V = Array.from({length: n}, (_, i) => Array.from({length: n}, (_, j) => (i === j ? 1 : 0)));
for (let sweep = 0; sweep < 30; sweep++) {
let off = 0; for (let p = 0; p < n; p++) for (let q = p + 1; q < n; q++) off += a[p][q] * a[p][q]; if (off < 1e-20) break;
for (let p = 0; p < n; p++) for (let q = p + 1; q < n; q++) {
if (Math.abs(a[p][q]) < 1e-30) continue;
const th = (a[q][q] - a[p][p]) / (2 * a[p][q]), t = Math.sign(th || 1) / (Math.abs(th) + Math.sqrt(th * th + 1)), c = 1 / Math.sqrt(t * t + 1), s = t * c;
for (let k = 0; k < n; k++) { const akp = a[k][p], akq = a[k][q]; a[k][p] = c * akp - s * akq; a[k][q] = s * akp + c * akq; }
for (let k = 0; k < n; k++) { const apk = a[p][k], aqk = a[q][k]; a[p][k] = c * apk - s * aqk; a[q][k] = s * apk + c * aqk; }
for (let k = 0; k < n; k++) { const vkp = V[k][p], vkq = V[k][q]; V[k][p] = c * vkp - s * vkq; V[k][q] = s * vkp + c * vkq; }
}
}
return {B: V, d: a.map((r, i) => Math.max(1e-20, r[i]))};
}
class CMA {
constructor(n, sigma, seed) {
this.n = n; this.sigma = sigma; this.R = rng(seed || 1);
const lam = this.lam = 4 + Math.floor(3 * Math.log(n)), mu = this.mu = Math.floor(lam / 2);
let w = []; for (let i = 0; i < mu; i++) w.push(Math.log(mu + .5) - Math.log(i + 1)); const sw = w.reduce((a, b) => a + b, 0); w = w.map(v => v / sw); this.w = w;
const mueff = this.mueff = 1 / w.reduce((a, b) => a + b * b, 0);
this.cc = (4 + mueff / n) / (n + 4 + 2 * mueff / n); this.cs = (mueff + 2) / (n + mueff + 5);
this.c1 = 2 / ((n + 1.3) ** 2 + mueff); this.cmu = Math.min(1 - this.c1, 2 * (mueff - 2 + 1 / mueff) / ((n + 2) ** 2 + mueff));
this.damps = 1 + 2 * Math.max(0, Math.sqrt((mueff - 1) / (n + 1)) - 1) + this.cs; this.chiN = Math.sqrt(n) * (1 - 1 / (4 * n) + 1 / (21 * n * n));
this.m = new Array(n).fill(0); this.pc = new Array(n).fill(0); this.ps = new Array(n).fill(0);
this.C = Array.from({length: n}, (_, i) => Array.from({length: n}, (_, j) => (i === j ? 1 : 0))); this.B = this.C.map(r => r.slice()); this.D = new Array(n).fill(1);
this.gen = 0; this.best = {f: Infinity, z: this.m.slice()};
}
ask() {
const n = this.n, P = [];
for (let k = 0; k < this.lam; k++) {
const zz = Array.from({length: n}, () => this.R.n()), y = new Array(n).fill(0);
for (let i = 0; i < n; i++) { let s = 0; for (let j = 0; j < n; j++) s += this.B[i][j] * this.D[j] * zz[j]; y[i] = s; }
P.push({y, z: this.m.map((v, i) => v + this.sigma * y[i])});
}
return P;
}
tell(P, F) {
const n = this.n, idx = F.map((v, i) => i).sort((a, b) => F[a] - F[b]);
if (F[idx[0]] < this.best.f) this.best = {f: F[idx[0]], z: P[idx[0]].z.slice()};
const yw = new Array(n).fill(0); for (let i = 0; i < this.mu; i++) for (let j = 0; j < n; j++) yw[j] += this.w[i] * P[idx[i]].y[j];
for (let j = 0; j < n; j++) this.m[j] += this.sigma * yw[j];
const t = new Array(n).fill(0); for (let i = 0; i < n; i++) { let s = 0; for (let j = 0; j < n; j++) s += this.B[j][i] * yw[j]; t[i] = s / this.D[i]; }
const cinv = new Array(n).fill(0); for (let i = 0; i < n; i++) { let s = 0; for (let j = 0; j < n; j++) s += this.B[i][j] * t[j]; cinv[i] = s; }
const a = Math.sqrt(this.cs * (2 - this.cs) * this.mueff); for (let i = 0; i < n; i++) this.ps[i] = (1 - this.cs) * this.ps[i] + a * cinv[i];
const psn = Math.sqrt(this.ps.reduce((s, v) => s + v * v, 0)); this.gen++;
const hsig = psn / Math.sqrt(1 - Math.pow(1 - this.cs, 2 * this.gen)) / this.chiN < 1.4 + 2 / (n + 1) ? 1 : 0;
const b = Math.sqrt(this.cc * (2 - this.cc) * this.mueff); for (let i = 0; i < n; i++) this.pc[i] = (1 - this.cc) * this.pc[i] + hsig * b * yw[i];
for (let i = 0; i < n; i++) for (let j = 0; j <= i; j++) {
let r = 0; for (let k = 0; k < this.mu; k++) r += this.w[k] * P[idx[k]].y[i] * P[idx[k]].y[j];
const v = (1 - this.c1 - this.cmu) * this.C[i][j] + this.c1 * (this.pc[i] * this.pc[j] + (1 - hsig) * this.cc * (2 - this.cc) * this.C[i][j]) + this.cmu * r;
this.C[i][j] = v; this.C[j][i] = v;
}
this.sigma *= Math.exp(this.cs / this.damps * (psn / this.chiN - 1)); this.sigma = Math.min(this.sigma, 3);
const E = eigenSym(this.C, n); this.B = E.B; this.D = E.d.map(Math.sqrt);
}
}

}

function fitFiltEG(curve, FC, hold) {

const nm = nelderMead(err, best.p, [8, 10, 12, 12, 20, 12], 320, [[0, 127], [-63, 63], [0, 127], [0, 127], [0, 127], [0, 127]]);

}
function beatAnalysis(A, tF, frames, K, f0) {

for (let i = 0; i < n; i++) { const v = u[i]; s0++; s1 += v; s2 += v * v; s3 += v * v * v; s4 += v * v * v * v; y0 += y[i]; y1 += v * y[i]; y2 += v * v * y[i]; }

tracks.push({k, d, vr, mp});
}
let mpTot = 0; for (const x of tracks) mpTot += x.mp;
const fitK = (x, fb) => {
