if(r.arp)add(i,lk?{src:src||'clip',n,raw:src==='melody',key:key||null,vel,on:t,off:t+dur,lk}:{src:src||'clip',n,raw:src==='melody',key:key||null,vel,on:t,off:t+dur});
return r.hold;
}

function cut(i,t){const s=ST.get(i);if(!s)return;let ch=false;for(const x of s.iv)if(x.src!=='live'&&x.off>t){x.off=Math.max(x.on,t);ch=true;}s.brk=Math.max(s.brk,t);void ch;rewind(i,t);}
function cancelFrom(i,t){const s=ST.get(i);if(!s)return;const n0=s.iv.length;s.iv=s.iv.filter(x=>x.src==='live'||x.on<t-1e-6);if(s.iv.length!==n0)rewind(i,t);}


function carrier(eff,noRetune){
for(const i of[...ST.keys()]){
rewind(i,eff);plan(i,nowT(),LOOKAHEAD);
if(noRetune||!ST.has(i))continue;
const tr=S.tracks[i];if(!tr||tr.onChange==='hold')continue;
for(const v of VOICES)if(!v.dead&&v.track===i&&v.src==='arp'&&v.arp&&v.start<eff&&v.end>eff){
const x=v.arp.x,p=clamp((x.raw?x.n:outPitch(i,x.key,x.n,eff))+12*v.arp.o,0,127);if(p!==v.out)v.retune(p,eff);
}
}
}
function tick(now,h){
if(!ST.size)return;
const hz=typeof h==='number'&&h>0?h:LOOKAHEAD;
for(const i of[...ST.keys()]){
const s=ST.get(i);

if(s.iv.length>8){const q=query(s,now,true),keep=q?q.g.on:-Infinity;s.iv=s.iv.filter(x=>x.on>=keep-E_||x.off>now-.5);}
plan(i,now,hz);
}
}

function cur(i,t){const s=ST.get(i);if(!s)return null;let r=null;for(const x of s.hist){if(x.t<=t+1e-4)r=x;else break;}return r&&r.s>=0?{s:r.s,t:r.t,notes:r.vs.length}:null;}
function arpLog(i){const tr=S.tracks[i],k=tr&&tr.uid?tr.uid:'t'+i;return deep(LOG.get(k)||[]);}
function clearLog(i){if(i==null)LOG.clear();else{const tr=S.tracks[i];LOG.delete(tr&&tr.uid?tr.uid:'t'+i);}}
const info=i=>{const s=ST.get(i);return s?{grid:s.grid&&s.grid.k,next:s.next,held:s.iv.length,pc:s.pc,kOn:s.kOn}:null;};
const seedOf=i=>{const tr=S.tracks[i];return mix(fnv(String(S.id||'')+'|'+(tr&&tr.uid||i)),7);};
if(typeof window!=='undefined')window.ARP1K={gnNotes,gnLen,ST,A_,seedOf};
return{RES,TYP,arpPattern,seedOf,on,live,liveUp,clip,cut,cancelFrom,carrier,rewind,tick,plan,end,endAll,cur,arpLog,clearLog,info,busy:()=>ST.size>0,route};
})();



const BMARP=(function(){
'use strict';
const T=['off','1oct','dbl','2oct','alt','3oct','pong','pon2','pon3','rnd','rnd2','rnd3','min','maj','sus2','sus4','7th','min2','maj2','su22','su42','7th2','1o','2o','3o','5th','5th2','trp','trp2'];

const FIX={min:[0,3,7],maj:[0,4,7],sus2:[0,2,7],sus4:[0,5,7],'7th':[0,4,7,10],'1o':[0],'2o':[0,12],'3o':[0,12,24],'5th':[0,7],'5th2':[0,7,12,19]};
for(const k of['min','maj','sus2','sus4','7th'])FIX[k==='sus2'?'su22':k==='sus4'?'su42':k+'2']=FIX[k].concat(FIX[k].map(x=>x+12));
const md=(a,n)=>((a%n)+n)%n;
const mix=(h,c)=>{let x=(h^Math.imul(c|0,0x9e3779b1))>>>0;x^=x>>>16;x=Math.imul(x,0x85ebca6b)>>>0;x^=x>>>13;x=Math.imul(x,0xc2b2ae35)>>>0;x^=x>>>16;return x>>>0;};
const oct=(keys,n)=>{const U=[];for(let o=0;o<n;o++)for(const k of keys)U.push(k+12*o);return U;};
const pong=U=>(U.length<2?U:U.concat(U.slice(1,-1).reverse()));
const treppe=U=>{const T=U.slice(0,2);for(let j=0;j+2<U.length;j++)T.push(U[j],U[j+2]);return T;};

function seq(t,keys){
if(!keys.length)return[];
switch(t){
case'1oct':return keys.slice();
case'dbl':{const o=[];for(const k of keys)o.push(k,k);return o;}
case'2oct':return oct(keys,2);
case'3oct':return oct(keys,3);
case'alt':{const o=[];for(const k of keys)o.push(k,k+12);return o;}
case'pong':return pong(keys);
case'pon2':return pong(oct(keys,2));
case'pon3':return pong(oct(keys,3));
case'trp':return treppe(keys);
case'trp2':return treppe(oct(keys,2));
default:{const f=FIX[t];if(f){const last=keys[keys.length-1];return f.map(x=>last+x);}return[];}
}
}

function step(t,keys,c,seed){
if(!keys.length||t==='off')return null;
if(t==='rnd'||t==='rnd2'||t==='rnd3'){const U=oct(keys,t==='rnd'?1:t==='rnd2'?2:3);const x=U[mix(seed>>>0,c)%U.length];return x<=127?x:null;}
const s=seq(t,keys);if(!s.length)return null;const n=s[md(c,s.length)];return n<=127?n:null;
}

const DIV=[[.25,'1/4',1],[.5,'1/8',.5],[.75,'1/8T',1/3],[1,'1/16',.25],[1.5,'1/16T',1/6],[2,'1/32',.125]];
function div(spd,bpm,running){
spd=Math.max(20,Math.min(400,+spd||120));
if(!running)return{sec:60/spd/4,name:'frei',beats:spd?bpm/spd/4:0};
const r=spd/bpm;let best=DIV[0],bd=Infinity;for(const d of DIV){const x=Math.abs(Math.log(r/d[0]));if(x<bd-1e-12){bd=x;best=d;}}
return{sec:best[2]*60/bpm,name:best[1],beats:best[2]};
}
return{T,FIX,seq,step,div,stop:i=>{if(typeof ARPC!=='undefined')ARPC.end(i);}};
})();
if(typeof window!=='undefined')window.BMARP=BMARP;
